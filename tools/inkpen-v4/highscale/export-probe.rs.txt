    #[test]
    fn agent_v4_highscale_exports() {
        let receipt: Value = serde_json::from_slice(&std::fs::read(std::path::Path::new(&std::env::var("SDOCX_V4_HIGH_DIR").unwrap()).join("packed.json")).unwrap()).unwrap();
        let root = std::path::PathBuf::from(std::env::var("SDOCX_V4_HIGH_DIR").unwrap());
        let dir = root.join("exports");
        std::fs::create_dir_all(&dir).unwrap();
        for case in receipt["cases"].as_array().unwrap() {
            let name = case["name"].as_str().unwrap();
            let scale = case["scale"].as_u64().unwrap() as u32;
            let mut stroke = native_caller_stroke(&case["input"]);
            let rendering = stroke.rendering.as_mut().unwrap();
            rendering.advanced_settings = Some("4;".into());
            rendering.style.color_argb = Some(case["argb"].as_u64().unwrap() as u32);
            let page = crate::Page { uuid: "v4-highscale".into(), width: 128, height: 64, content_bbox: BoundingBox::default(), background_color: None, template: None, background: Default::default(), objects: vec![stroke.into()] };
            let document = crate::Document { pages: vec![page], metadata: Default::default() };
            for contour in [1, scale] {
                let options = crate::RenderOptions { inkpen_profile: Some(crate::InkPenRenderProfile { screen_width: 1920, screen_height: 1080 }), inkpen_v4_output_scale: Some(contour), ..crate::RenderOptions::default() };
                std::fs::write(dir.join(format!("{name}-{contour}.svg")), crate::render_document_svg(&document, &options).remove(0).svg).unwrap();
                std::fs::write(dir.join(format!("{name}-{contour}.pdf")), crate::render_document_pdf(&document, &options, &crate::PdfOptions::default()).unwrap()).unwrap();
            }
        }
    }
